let multiplier = 1.00; let running = false; let balance = 10000; function startRound() { multiplier = 1.00; running = true; let crashPoint = (Math.random() * 9 + 1.01).toFixed(2); document.getElementById('multi').style.color = '#00C853'; let interval = setInterval(() => { if(running) { multiplier += 0.03; document.getElementById('multi').innerText = multiplier.toFixed(2) + 'x'; if(multiplier > 2) document.getElementById('multi').style.color = '#FFD600'; if(multiplier > 5) document.getElementById('multi').style.color = '#FF3D00'; if(multiplier >= crashPoint) { running = false; clearInterval(interval); document.getElementById('multi').innerText = 'CRASHED @ ' + crashPoint + 'x'; document.getElementById('multi').style.color = '#FF3D00'; setTimeout(startRound, 4000); } } }, 50); } function bet(betNum) { if(!running) return alert('Wait for next round'); let amount = parseFloat(document.getElementById('bet'+betNum).value); if(amount <= 0 || amount > balance) return alert('Invalid amount'); balance -= amount; document.getElementById('balance').innerText = 'Balance: ' + balance.toFixed(2) + ' GHS'; document.getElementById('btn'+betNum).innerText = 'CASH OUT'; document.getElementById('btn'+betNum).style.background = '#00C853'; document.getElementById('btn'+betNum).onclick = () => cashout(betNum, amount); } function cashout(betNum, amount) { if(!running) return; let win = amount * multiplier; balance += win; document.getElementById('balance').innerText = 'Balance: ' + balance.toFixed(2) + ' GHS'; alert('Won ' + win.toFixed(2) + ' GHS'); document.getElementById('btn'+betNum).innerText = 'Bet'; document.getElementById('btn'+betNum).style.background = '#2979FF'; document.getElementById('btn'+betNum).onclick = () => bet(betNum); } startRound();
